function(real_library target kind path)
  if(NOT EXISTS "${path}")
    message(FATAL_ERROR "Missing actual library for ${target}: ${path}")
  endif()
  add_library("${target}" ${kind} IMPORTED GLOBAL)
  set_property(TARGET "${target}" PROPERTY IMPORTED_LOCATION "${path}")
endfunction()

function(resolve_first out_var)
  foreach(candidate IN LISTS ARGN)
    if(EXISTS "${candidate}")
      set("${out_var}" "${candidate}" PARENT_SCOPE)
      return()
    endif()
  endforeach()
  message(FATAL_ERROR "None of the candidate files exist for ${out_var}: ${ARGN}")
endfunction()

# Satisfy upstream find_package(JNI REQUIRED) when using openjdk-17-jdk-headless
# (libtakenginejni overrides LINK_LIBRARIES explicitly and does not use jawt).
find_file(_takx_jawt_header NAMES jawt.h PATHS "$ENV{JAVA_HOME}/include" NO_DEFAULT_PATH)
if(NOT _takx_jawt_header)
  find_library(_takx_jvm_lib NAMES jvm
    PATHS "$ENV{JAVA_HOME}/lib/server" "$ENV{JAVA_HOME}/jre/lib/amd64/server"
    NO_DEFAULT_PATH REQUIRED)
  set(JAVA_AWT_INCLUDE_PATH "$ENV{JAVA_HOME}/include" CACHE PATH "Headless JDK include fallback" FORCE)
  set(JAVA_AWT_LIBRARY "${_takx_jvm_lib}" CACHE FILEPATH "Headless JDK JNI fallback" FORCE)
endif()

      INCLUDE_DIRECTORIES "${GLES_EXTRA}")
